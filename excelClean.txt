<#
.SYNOPSIS
    Cleans SkanExcelAddIn VSTO per-user deployment for ALL users on the machine.

.DESCRIPTION
    - Detects users BEFORE cleanup (so VSTOInstaller doesn't hide the state)
    - If Skan is installed at machine level (MSI), cleans ALL user profiles
    - Skips users only if Skan is NOT installed at machine level AND user has no state
    - No hardcoded GUIDs, product codes, or version-specific hashes
    - Sets LoadBehavior=3 so Excel auto-loads on next start
    - Runs silently with no user prompts
    - Logs all actions to C:\Windows\SysWOW64\PKG_LOGS for audit
    - Requires Admin privileges

.NOTES
    Version: 2.0
    Date:    2026-04-07
#>

param([switch]$WhatIf)

$ErrorActionPreference = 'Stop'

$scriptName = "SkanExcelAddIn_PerUser_Cleanup"
$logDir = "C:\Windows\SysWOW64\PKG_LOGS"
$timestamp = Get-Date -Format 'yyyyMMdd_HHmmss'
$logFile = "$logDir\${scriptName}_${timestamp}.log"

function Write-Log {
    param([string]$Message, [string]$Level = "INFO")
    $ts = Get-Date -Format 'yyyy-MM-dd HH:mm:ss'
    $line = "$ts [$Level] $Message"
    try { if (Test-Path (Split-Path $logFile -Parent)) { Add-Content -Path $logFile -Value $line -ErrorAction SilentlyContinue } } catch {}
    switch ($Level) {
        "ERROR" { Write-Host "  $line" -ForegroundColor Red }
        "WARN"  { Write-Host "  $line" -ForegroundColor Yellow }
        "OK"    { Write-Host "  $line" -ForegroundColor Green }
        default { Write-Host "  $line" }
    }
}

function Remove-RegistryKey {
    param([string]$Path)
    if ($WhatIf) { Write-Log "  [WHATIF] Would delete: $Path"; return $false }
    try {
        if (Test-Path $Path) { Remove-Item $Path -Recurse -Force -ErrorAction Stop; return $true }
    } catch { Write-Log "  Failed to delete: $Path -- $($_.Exception.Message)" "WARN" }
    return $false
}

function Unload-Hive {
    param([string]$HivePath)
    [gc]::Collect(); [gc]::WaitForPendingFinalizers(); Start-Sleep -Milliseconds 500
    for ($i = 1; $i -le 3; $i++) {
        & reg.exe UNLOAD $HivePath 2>$null | Out-Null
        if ($LASTEXITCODE -eq 0) { return $true }
        [gc]::Collect(); Start-Sleep -Seconds 2
    }
    Write-Log "  Failed to unload hive $HivePath after 3 attempts" "WARN"
    return $false
}

# =====================================================================
# PRE-FLIGHT
# =====================================================================

$isAdmin = ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]'Administrator')
if (-not $isAdmin) { Write-Log "Requires Administrator. Exiting." "ERROR"; exit 1 }
if (-not (Test-Path $logDir)) { New-Item -ItemType Directory -Path $logDir -Force -ErrorAction SilentlyContinue | Out-Null }

Write-Log "========== $scriptName v2.0 started =========="
Write-Log "Machine: $env:COMPUTERNAME | User: $env:USERDOMAIN\$env:USERNAME"
if ($WhatIf) { Write-Log "*** DRY RUN -- no changes ***" "WARN" }

# =====================================================================
# STEP 1: DETECT -- check machine-level install and build user list FIRST
# (Must happen BEFORE VSTOInstaller runs, which can erase the evidence)
# =====================================================================

Write-Log "--- STEP 1: Detect SkanExcelAddIn state ---"

$hklmAddin = Get-ItemProperty "HKLM:\SOFTWARE\Microsoft\Office\Excel\Addins\SkanExcelAddIn" -ErrorAction SilentlyContinue
$dllExists = Test-Path "C:\Program Files\Skan\VA\SkanExcelAddin\SkanExcelAddIn.dll"
$skanMachineLevel = ($null -ne $hklmAddin) -or $dllExists

if ($skanMachineLevel) {
    Write-Log "Skan installed at machine level. Will process ALL user profiles." "OK"
} else {
    Write-Log "Skan NOT at machine level. Will only clean users with leftover per-user state."
}

$profileListKey = 'HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion\ProfileList'
$sidEntries = Get-ChildItem -Path $profileListKey -ErrorAction SilentlyContinue
$usersToClean = @()
$totalUsers = 0
$skippedUsers = 0

$skipSidPrefixes = @('S-1-5-18', 'S-1-5-19', 'S-1-5-20')
$skipUserNames = @('defaultuser100000', 'Default', 'Public', 'All Users', 'Default User')

foreach ($sidEntry in $sidEntries) {
    $sid = $sidEntry.PSChildName

    # Include: S-1-5-21-* (local/AD) and S-1-12-1-* (Azure AD)
    # Skip: system SIDs (LocalSystem, LocalService, NetworkService)
    $isUserSid = ($sid -like 'S-1-5-21-*') -or ($sid -like 'S-1-12-1-*')
    $isSystemSid = $skipSidPrefixes | Where-Object { $sid -eq $_ }
    if (-not $isUserSid -or $isSystemSid) { continue }

    $profPathRaw = (Get-ItemProperty -Path $sidEntry.PSPath -Name 'ProfileImagePath' -ErrorAction SilentlyContinue).ProfileImagePath
    if ([string]::IsNullOrWhiteSpace($profPathRaw)) { continue }
    $profPath = [Environment]::ExpandEnvironmentVariables($profPathRaw)
    if (-not (Test-Path $profPath)) { continue }

    $userName = Split-Path $profPath -Leaf
    if ($userName -in $skipUserNames) { continue }

    $totalUsers++

    $shouldClean = $false

    if ($skanMachineLevel) {
        $shouldClean = $true
    } else {
        # Check 1: skan* folders in Apps\2.0
        $userApps20 = Join-Path $profPath "AppData\Local\Apps\2.0"
        if (Test-Path $userApps20 -ErrorAction SilentlyContinue) {
            $sc = Get-ChildItem -LiteralPath $userApps20 -Directory -Recurse -Filter "skan*" -ErrorAction SilentlyContinue
            if ($sc -and $sc.Count -gt 0) { $shouldClean = $true }
        }

        # Check 2: HKCU registry (for logged-in users whose hive is loaded)
        if (-not $shouldClean) {
            if (Test-Path "Registry::HKEY_USERS\$sid" -ErrorAction SilentlyContinue) {
                $hkuCheck = "Registry::HKEY_USERS\$sid"
                # Excel addin key
                if (Test-Path "$hkuCheck\Software\Microsoft\Office\Excel\Addins\SkanExcelAddIn" -ErrorAction SilentlyContinue) { $shouldClean = $true }
                # SideBySide skan keys
                if (-not $shouldClean) {
                    $sbsCheck = "$hkuCheck\SOFTWARE\Classes\Software\Microsoft\Windows\CurrentVersion\Deployment\SideBySide\2.0"
                    if (Test-Path $sbsCheck -ErrorAction SilentlyContinue) {
                        $sr = Get-ChildItem $sbsCheck -Recurse -ErrorAction SilentlyContinue | Where-Object { $_.PSChildName -like '*skan*' } | Select-Object -First 1
                        if ($sr) { $shouldClean = $true }
                    }
                }
                # Uninstall entries with Skan in DisplayName
                if (-not $shouldClean) {
                    $uninstCheck = "$hkuCheck\Software\Microsoft\Windows\CurrentVersion\Uninstall"
                    if (Test-Path $uninstCheck -ErrorAction SilentlyContinue) {
                        Get-ChildItem $uninstCheck -ErrorAction SilentlyContinue | ForEach-Object {
                            $p = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
                            if ($p.DisplayName -like '*Skan*' -or $p.DisplayName -like '*skan*') { $shouldClean = $true }
                        }
                    }
                }
            }
        }
    }

    if ($shouldClean) {
        $usersToClean += [PSCustomObject]@{ SID = $sid; ProfilePath = $profPath; UserName = $userName }
        Write-Log "  Will clean: $userName ($sid)"
    } else {
        $skippedUsers++
    }
}

Write-Log "Total profiles: $totalUsers | To clean: $($usersToClean.Count) | To skip: $skippedUsers"

# =====================================================================
# STEP 2: KILL processes
# =====================================================================

Write-Log "--- STEP 2: Kill Excel and SkanWatchDog ---"
$ErrorActionPreference = 'SilentlyContinue'
& taskkill /IM "EXCEL.EXE" /F /T *>&1 | Out-Null
& taskkill /IM "SkanWatchDog.exe" /F /T *>&1 | Out-Null
$ErrorActionPreference = 'Stop'
Start-Sleep -Seconds 3
Write-Log "Done." "OK"

# =====================================================================
# STEP 3: VSTOInstaller /Uninstall (machine level)
# =====================================================================

Write-Log "--- STEP 3: VSTOInstaller /Uninstall ---"
$vstoExe = "C:\Program Files\Common Files\Microsoft Shared\VSTO\10.0\VSTOInstaller.exe"
@(
    "C:\Program Files\Skan\VA\SkanExcelAddin\SkanExcelAddIn.vsto",
    "C:\Program Files\Skan\VA\SkanOutlookAddin\Skan.Outlook.Addin.vsto"
) | ForEach-Object {
    if ((Test-Path $vstoExe) -and (Test-Path $_)) {
        $name = [System.IO.Path]::GetFileNameWithoutExtension($_)
        try {
            $proc = Start-Process -FilePath $vstoExe -ArgumentList "/Uninstall `"$_`" /SILENT" -Wait -PassThru -WindowStyle Hidden
            Write-Log "  $name exit code: $($proc.ExitCode)"
        } catch {
            Write-Log "  $name failed: $($_.Exception.Message)" "WARN"
        }
    }
}

# =====================================================================
# STEP 4: CLEAN each user
# =====================================================================

Write-Log "--- STEP 4: Clean per-user state ---"
$cleanedUsers = 0

foreach ($userInfo in $usersToClean) {
    $sid = $userInfo.SID
    $profPath = $userInfo.ProfilePath
    $userName = $userInfo.UserName

    try {
        Write-Log "  Processing: $userName"

        $userApps20 = Join-Path $profPath "AppData\Local\Apps\2.0"
        $hiveLoaded = Test-Path "Registry::HKEY_USERS\$sid"
        $weLoaded = $false

        if (-not $hiveLoaded) {
            $ntUserDat = Join-Path $profPath 'NTUSER.DAT'
            if (Test-Path -LiteralPath $ntUserDat) {
                & reg.exe LOAD "HKU\$sid" "`"$ntUserDat`"" 2>$null | Out-Null
                if ($LASTEXITCODE -eq 0) { $weLoaded = $true; $hiveLoaded = $true }
            }
        }

        # --- Apps\2.0: find skan*, delete entire parent store ---
        if ((Test-Path $userApps20) -and -not $WhatIf) {
            $skanDirs = Get-ChildItem -LiteralPath $userApps20 -Directory -Recurse -Filter "skan*" -ErrorAction SilentlyContinue
            if ($skanDirs -and $skanDirs.Count -gt 0) {
                $storeParents = @()
                foreach ($sd in $skanDirs) {
                    $parent = Split-Path $sd.FullName -Parent
                    if ($storeParents -notcontains $parent) { $storeParents += $parent }
                }
                foreach ($store in $storeParents) {
                    Get-ChildItem -LiteralPath $store -ErrorAction SilentlyContinue | ForEach-Object {
                        try { Remove-Item -LiteralPath $_.FullName -Recurse -Force -ErrorAction Stop }
                        catch { Write-Log "    Failed: $($_.Name) -- $($_.Exception.Message)" "WARN" }
                    }
                }
                $dataPath = Join-Path $userApps20 "Data"
                if (Test-Path $dataPath) {
                    $dataSkan = Get-ChildItem -LiteralPath $dataPath -Directory -Recurse -Filter "skan*" -ErrorAction SilentlyContinue
                    if ($dataSkan) {
                        $dataParents = @()
                        foreach ($ds in $dataSkan) { $dp = Split-Path $ds.FullName -Parent; if ($dataParents -notcontains $dp) { $dataParents += $dp } }
                        foreach ($dp in $dataParents) { Remove-Item -LiteralPath $dp -Recurse -Force -ErrorAction SilentlyContinue }
                    }
                }
                @("manifests.dat", "data.dat") | ForEach-Object {
                    $mf = Join-Path $userApps20 $_; if (Test-Path $mf) { Remove-Item $mf -Force -ErrorAction SilentlyContinue }
                }
                Write-Log "    Apps\2.0 cleaned" "OK"
            }
        }

        # --- HKU registry ---
        if ($hiveLoaded) {
            $hkuBase = "Registry::HKEY_USERS\$sid"

            @(
                "$hkuBase\Software\Microsoft\Office\Excel\Addins\SkanExcelAddIn",
                "$hkuBase\Software\Microsoft\Office\Excel\AddinsData\SkanExcelAddIn",
                "$hkuBase\SOFTWARE\Microsoft\Office\Outlook\Addins\Skan.Outlook.Addin",
                "$hkuBase\SOFTWARE\Microsoft\Office\Outlook\AddinsData\Skan.Outlook.Addin"
            ) | ForEach-Object { Remove-RegistryKey $_ | Out-Null }

            @("15.0", "16.0") | ForEach-Object {
                Remove-RegistryKey "$hkuBase\Software\Microsoft\Office\$_\Excel\AddInLoadTimes" | Out-Null
            }

            $uninstPath = "$hkuBase\Software\Microsoft\Windows\CurrentVersion\Uninstall"
            if (Test-Path $uninstPath) {
                Get-ChildItem $uninstPath -ErrorAction SilentlyContinue | ForEach-Object {
                    try {
                        $props = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
                        if ($props.DisplayName -like '*Skan*' -or $props.DisplayName -like '*skan*' -or
                            $props.Publisher -like '*Skan*' -or $props.Publisher -like '*skan*') {
                            Remove-RegistryKey $_.PSPath | Out-Null
                        }
                    } catch {}
                }
            }

            foreach ($metaPath in @("$hkuBase\SOFTWARE\Microsoft\VSTA\Solutions", "$hkuBase\SOFTWARE\Microsoft\VSTO\SolutionMetadata")) {
                if (Test-Path $metaPath) {
                    Get-ChildItem $metaPath -ErrorAction SilentlyContinue | ForEach-Object {
                        try {
                            $isSkan = $false
                            $props = Get-ItemProperty $_.PSPath -ErrorAction SilentlyContinue
                            $props.PSObject.Properties | ForEach-Object {
                                if ($_.Value -is [string] -and ($_.Value -like '*Skan*' -or $_.Value -like '*skan*')) { $isSkan = $true }
                            }
                            if ($isSkan) { Remove-RegistryKey $_.PSPath | Out-Null }
                        } catch {}
                    }
                }
            }

            $sbsPath = "$hkuBase\SOFTWARE\Classes\Software\Microsoft\Windows\CurrentVersion\Deployment\SideBySide\2.0"
            if (Test-Path $sbsPath) {
                $skanSbs = Get-ChildItem $sbsPath -Recurse -ErrorAction SilentlyContinue |
                    Where-Object { $_.PSChildName -like '*skan*' } | Select-Object -First 1
                if ($skanSbs) { Remove-RegistryKey $sbsPath | Out-Null }
            }

            @("15.0", "16.0") | ForEach-Object {
                $resBase = "$hkuBase\Software\Microsoft\Office\$_\Excel\Resiliency"
                @("DisabledItems", "CrashingAddinList", "DoNotDisableAddinList", "NotificationReminderAddinData") | ForEach-Object {
                    Remove-RegistryKey "$resBase\$_" | Out-Null
                }
            }

            Write-Log "    Registry cleaned" "OK"

            # --- Set LoadBehavior=3 ---
            if ($hklmAddin -and -not $WhatIf) {
                $addinPath = "$hkuBase\Software\Microsoft\Office\Excel\Addins\SkanExcelAddIn"
                try {
                    if (-not (Test-Path $addinPath)) { New-Item -Path $addinPath -Force -ErrorAction Stop | Out-Null }
                    New-ItemProperty -Path $addinPath -Name "LoadBehavior" -PropertyType DWord -Value 3 -Force -ErrorAction Stop | Out-Null
                    New-ItemProperty -Path $addinPath -Name "FriendlyName" -PropertyType String -Value "SkanExcelAddIn" -Force -ErrorAction Stop | Out-Null
                    New-ItemProperty -Path $addinPath -Name "Description" -PropertyType String -Value "SkanExcelAddIn" -Force -ErrorAction Stop | Out-Null
                    if ($hklmAddin.Manifest) {
                        New-ItemProperty -Path $addinPath -Name "Manifest" -PropertyType String -Value $hklmAddin.Manifest -Force -ErrorAction Stop | Out-Null
                    }
                    Write-Log "    LoadBehavior=3 set" "OK"
                } catch {
                    Write-Log "    Failed to set LoadBehavior: $($_.Exception.Message)" "WARN"
                }
            }
        }

        if ($weLoaded) { Unload-Hive "HKU\$sid" | Out-Null }
        $cleanedUsers++
        Write-Log "  [DONE] $userName" "OK"

    } catch {
        Write-Log "  Error for $userName : $($_.Exception.Message)" "ERROR"
        if ($weLoaded) { Unload-Hive "HKU\$sid" | Out-Null }
    }
}

# =====================================================================
# SUMMARY
# =====================================================================

Write-Log ""
Write-Log "========== $scriptName completed =========="
Write-Log "Total profiles: $totalUsers | Cleaned: $cleanedUsers | Skipped: $skippedUsers"
Write-Log "Log: $logFile"
if ($cleanedUsers -gt 0 -and $hklmAddin) {
    Write-Log "LoadBehavior=3 set. Excel will auto-install SkanExcelAddIn on next open." "OK"
} elseif ($cleanedUsers -eq 0) {
    Write-Log "No users needed cleaning." "OK"
}
